  {\vspace{1em}\large\bfseries} % تنسيق الخط للفصل
  {\contentslabel{2.5em}} % عرض مساحة الرقم

  {\hfill\contentspage} % حشو السطر ورقم الصفحة بدون نقاط

\titlecontents{section}
  [5.5em] % المسافة البادئة للقسم (2.5 + 3.0)
  {\vspace{0.2em}}
  {\contentslabel{3em}} % عرض مساحة الرقم
  {\hspace*{-3em}}
  {\titlerule*[0.5pc]{.}\contentspage} % نقاط تصل لرقم الصفحة

\titlecontents{subsection}
  [9em] % المسافة البادئة للقسم الفرعي (5.5 + 3.5)
  {\vspace{0.1em}}
  {\contentslabel{3.5em}} % عرض مساحة الرقم
  {\hspace*{-3.5em}}
  {\titlerule*[0.5pc]{.}\contentspage}
